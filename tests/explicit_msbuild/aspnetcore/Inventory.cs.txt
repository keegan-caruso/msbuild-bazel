using System.Text.Json;
using Microsoft.Build.Evaluation;
using Microsoft.Build.Graph;
using NuGet.Frameworks;
var sdk = Environment.GetEnvironmentVariable("RULES_MSBUILD_DOTNET_ROOT")! + "/sdk/10.0.400";
System.Runtime.Loader.AssemblyLoadContext.Default.Resolving += (c,n) => File.Exists(sdk+"/"+n.Name+".dll") ? c.LoadFromAssemblyPath(sdk+"/"+n.Name+".dll") : null;
Environment.SetEnvironmentVariable("MSBUILD_EXE_PATH", sdk+"/MSBuild.dll");
Environment.SetEnvironmentVariable("MSBuildSDKsPath", sdk+"/Sdks");
Scan(Path.GetFullPath(args[0]), args[1], args[2]);
static void Scan(string root, string selection, string output)
{
    var entries = JsonDocument.Parse(File.ReadAllText(selection)).RootElement.GetProperty("entries");
    using var collection = new ProjectCollection();
    var graph = new ProjectGraph(entries.EnumerateArray().Select(e => new ProjectGraphEntryPoint(Path.Combine(root,e.GetString()!), new Dictionary<string,string>{{"Configuration","Release"}})), collection, null);
    var rows = graph.ProjectNodes.Select(n => {
        var p=n.ProjectInstance;
        object[] Items(string name) => p.GetItems(name).Select(i => (object)new {include=i.EvaluatedInclude,metadata=i.Metadata.ToDictionary(m=>m.Name,m=>m.EvaluatedValue)}).ToArray();
        return new {project=Path.GetRelativePath(root,p.FullPath),framework=p.GetPropertyValue("TargetFramework"),globals=p.GlobalProperties,assets=p.GetPropertyValue("ProjectAssetsFile"),
            properties=new[]{"TargetFrameworks","LangVersion","Nullable","AssemblyName","RootNamespace","AllowUnsafeBlocks","AssemblyOriginatorKeyFile","OutputType","UseAppHost","IsTestProject","ExcludeFromBuild","EnableDefaultCompileItems","IntermediateOutputPath","OutputPath"}.ToDictionary(x=>x,x=>p.GetPropertyValue(x)),
            imports=p.ImportPaths, references=Items("ProjectReference"),packages=Items("PackageReference"),bareReferences=Items("Reference"),frameworks=Items("FrameworkReference"),compile=Items("Compile"),
            selectedEdges=n.ProjectReferences.Where(d=>d.ProjectInstance.GetPropertyValue("TargetFramework").Length>0).GroupBy(d=>d.ProjectInstance.FullPath).Select(g=>new {project=Path.GetRelativePath(root,g.Key),framework=p.GetPropertyValue("TargetFramework").Length == 0 ? "" : new FrameworkReducer().GetNearest(NuGetFramework.ParseFolder(p.GetPropertyValue("TargetFramework")),g.Select(d=>NuGetFramework.ParseFolder(d.ProjectInstance.GetPropertyValue("TargetFramework"))))?.GetShortFolderName()}).ToArray(),
            edges=n.ProjectReferences.Select(d=>new{project=Path.GetRelativePath(root,d.ProjectInstance.FullPath),framework=d.ProjectInstance.GetPropertyValue("TargetFramework"),globals=d.ProjectInstance.GlobalProperties}).ToArray(),
            items=new[]{"EmbeddedResource","Content","None","AdditionalFiles","GlobalAnalyzerConfigFiles","EditorConfigFiles"}.ToDictionary(t=>t,t=>Items(t))};
    }).ToArray();
    File.WriteAllText(output,JsonSerializer.Serialize(rows,new JsonSerializerOptions{WriteIndented=true}));
    Console.WriteLine($"Evaluated {rows.Length} configured graph nodes, {rows.Select(r=>r.project).Distinct().Count()} project files, {rows.Count(r=>r.framework.Length>0)} framework nodes");
}
