using System.Text.Json;
using Microsoft.Build.Evaluation;

var sdk = Path.GetFullPath(args[0]);
System.Runtime.Loader.AssemblyLoadContext.Default.Resolving += (context, name) => File.Exists(Path.Combine(sdk, name.Name + ".dll")) ? context.LoadFromAssemblyPath(Path.Combine(sdk, name.Name + ".dll")) : null;
Environment.SetEnvironmentVariable("MSBuildEnableWorkloadResolver", "false");
Environment.SetEnvironmentVariable("MSBUILD_EXE_PATH", Path.Combine(sdk, "MSBuild.dll"));
Environment.SetEnvironmentVariable("MSBuildSDKsPath", Path.Combine(sdk, "Sdks"));
Inventory();
void Inventory()
{
    var root = Path.GetFullPath(args[1]);
    string Logical(string path) => path.StartsWith(root + "/", StringComparison.Ordinal) ? Path.GetRelativePath(root, path) : path.StartsWith(sdk + "/", StringComparison.Ordinal) ? "<sdk>/" + Path.GetRelativePath(sdk, path) : path.Contains("/packages/", StringComparison.Ordinal) ? ".nuget/packages/" + path.Split("/packages/", 2)[1] : "<external>/" + Path.GetFileName(path);
    var result = new List<object>();
    foreach (var entry in args.Skip(3))
    {
        try
        {
            using var collection = new ProjectCollection();
            var globals = new Dictionary<string, string> { ["Configuration"] = "Release", ["Platform"] = "AnyCPU", ["ImportProjectExtensionProps"] = "false", ["ImportProjectExtensionTargets"] = "false" };
            foreach (var pair in JsonSerializer.Deserialize<Dictionary<string, string>>(Environment.GetEnvironmentVariable("RULES_MSBUILD_INVENTORY_PROPERTIES") ?? "{}")!) globals[pair.Key] = pair.Value;
            var outer = new Project(Path.Combine(root, entry), globals, null, collection);
            var frameworks = outer.GetPropertyValue("TargetFrameworks");
            if (frameworks.Length == 0) frameworks = outer.GetPropertyValue("TargetFramework");
            foreach (var framework in frameworks.Split(';', StringSplitOptions.RemoveEmptyEntries))
            {
                globals["TargetFramework"] = framework;
                var project = new Project(Path.Combine(root, entry), globals, null, collection);
                var documents = project.Imports.Select(i => i.ImportedProject).Append(project.Xml).Where(d => !d.FullPath.StartsWith(sdk + "/", StringComparison.Ordinal)).DistinctBy(d => d.FullPath).ToArray();
                result.Add(new {
                    project = entry, framework,
                    assets = project.GetPropertyValue("ProjectAssetsFile"),
                    items = project.AllEvaluatedItems.Where(i => !i.Xml.ContainingProject.FullPath.StartsWith(sdk + "/", StringComparison.Ordinal)).Select(i => new { type = i.ItemType, include = i.EvaluatedInclude, metadata = i.Metadata.ToDictionary(m => m.Name, m => m.EvaluatedValue) }).ToArray(),
                    customItems = project.AllEvaluatedItems.Where(i => !i.Xml.ContainingProject.FullPath.StartsWith(sdk + "/", StringComparison.Ordinal)).GroupBy(i => i.ItemType).OrderBy(g => g.Key).ToDictionary(g => g.Key, g => g.Count()),
                    documents = documents.Select(d => new { path = Logical(d.FullPath), sha256 = Convert.ToHexStringLower(System.Security.Cryptography.SHA256.HashData(File.ReadAllBytes(d.FullPath))), targets = d.Targets.Select(t => t.Name).ToArray(), tasks = d.UsingTasks.Select(t => t.TaskName).ToArray() }).ToArray(),
                    references = project.GetItems("Reference").Select(i => i.EvaluatedInclude).ToArray(),
                    projects = project.GetItems("ProjectReference").Select(i => Logical(i.GetMetadataValue("FullPath"))).ToArray(),
                    packages = project.GetItems("PackageReference").Select(i => new { id = i.EvaluatedInclude, version = i.GetMetadataValue("Version"), versionOverride = i.GetMetadataValue("VersionOverride") }).ToArray(),
                    missingSources = project.GetItems("Compile").Where(i => !File.Exists(i.GetMetadataValue("FullPath"))).Select(i => Logical(i.GetMetadataValue("FullPath"))).ToArray()
                });
            }
        }
        catch (Exception error) { result.Add(new { project = entry, error = error.Message.Replace(root, "<checkout>").Replace(sdk, "<sdk>") }); }
    }
    File.WriteAllText(args[2], JsonSerializer.Serialize(result, new JsonSerializerOptions { WriteIndented = true }));
}
