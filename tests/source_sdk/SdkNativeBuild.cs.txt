#nullable enable
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Formats.Tar;
using System.IO;
using System.IO.Compression;
using System.Linq;

// Qualification adapter: all native programs, headers and sources are declared
// archives. The nested namespace exposes only those inputs and writable scratch.
public sealed class NativeBuild
{
    public string Sandbox { get; set; } = null!;
    public string Toolchain { get; set; } = null!;
    public string Sources { get; set; } = null!;
    public string? Bootstrap { get; set; }
    public string? ScriptFile { get; set; }
    public string? SdkArchive { get; set; }
    public string? SdkLayout { get; set; }
    public string[] Dependencies { get; set; } = [];
    public string Script { get; set; } = null!;
    public string Output { get; set; } = null!;
    public string[] Products { get; set; } = [];
    public static void Main(string[] args)
    {
        string? bootstrap = null;
        string? scriptFile = null;
        string? sdkArchive = null;
        string? sdkLayout = null;
        var dependencies = new List<string>();
        while (args.Length >= 2 && args[^2] is "--bootstrap" or "--dependency" or "--script-file" or "--sdk-archive" or "--sdk-layout")
        {
            var path = args[^1];
            switch (args[^2])
            {
                case "--bootstrap": bootstrap = Path.GetFullPath(path); break;
                case "--script-file": scriptFile = Path.GetFullPath(path); break;
                case "--dependency": dependencies.Add(Path.GetFullPath(path)); break;
                case "--sdk-archive": sdkArchive = path; break;
                case "--sdk-layout": sdkLayout = Path.GetFullPath(path); break;
            }
            args = args[..^2];
        }
        dependencies.Reverse();
        if (args.Length < 6) throw new ArgumentException("Expected sandbox, toolchain, sources, script, archive and product outputs");
        if ((sdkArchive == null) != (sdkLayout == null)) throw new ArgumentException("SDK archive and layout must be supplied together");
        new NativeBuild { Bootstrap = bootstrap, ScriptFile = scriptFile, SdkArchive = sdkArchive, SdkLayout = sdkLayout, Dependencies = dependencies.ToArray(), Sandbox = Path.GetFullPath(args[0]), Toolchain = Path.GetFullPath(args[1]), Sources = Path.GetFullPath(args[2]), Script = args[3], Output = Path.GetFullPath(args[4]), Products = args.Skip(5).Select(Path.GetFullPath).ToArray() }.Execute();
    }
    public bool Execute()
    {
        var scratch = Path.Combine(Path.GetDirectoryName(Output)!, "native-scratch");
        var root = Path.Combine(scratch, "root");
        var source = Path.Combine(scratch, "source");
        if (Directory.Exists(scratch)) Directory.Delete(scratch, true);
        Directory.CreateDirectory(root);
        Directory.CreateDirectory(source);
        try
        {
            TarFile.ExtractToDirectory(Toolchain, root, false);
            TarFile.ExtractToDirectory(Sources, source, false);
            if (Bootstrap != null) TarFile.ExtractToDirectory(Bootstrap, source, false);
            if (ScriptFile != null)
            {
                if (Script != "/source/build-native.sh") throw new ArgumentException("Script override requires /source/build-native.sh");
                File.Copy(ScriptFile, Path.Combine(source, "build-native.sh"), true);
            }
            var dependencyRoot = Path.Combine(scratch, "dependencies");
            if (Dependencies.Length != 0)
            {
                Directory.CreateDirectory(dependencyRoot);
                foreach (var dependency in Dependencies) TarFile.ExtractToDirectory(dependency, dependencyRoot, false);
            }
            var start = new ProcessStartInfo(Sandbox) { RedirectStandardOutput = true, RedirectStandardError = true };
            foreach (var arg in new[] { "--die-with-parent", "--unshare-all", "--new-session", "--cap-drop", "ALL",
                "--ro-bind", root, "/", "--bind", source, "/source", "--proc", "/proc", "--dev", "/dev",
                "--tmpfs", "/tmp" })
                start.ArgumentList.Add(arg);
            if (Dependencies.Length != 0)
                foreach (var arg in new[] { "--ro-bind", dependencyRoot, "/tmp/component-inputs" })
                    start.ArgumentList.Add(arg);
            foreach (var arg in new[] { "--chdir", "/source", "--clearenv", "--setenv", "PATH", "/usr/bin:/bin",
                "--setenv", "HOME", "/tmp", "--setenv", "LANG", "C.UTF-8", "--", "/bin/bash", Script })
                start.ArgumentList.Add(arg);
            using var child = Process.Start(start)!;
            child.OutputDataReceived += (_, e) => { if (e.Data != null) Console.WriteLine(e.Data); };
            child.ErrorDataReceived += (_, e) => { if (e.Data != null) Console.WriteLine(e.Data); };
            child.BeginOutputReadLine(); child.BeginErrorReadLine(); child.WaitForExit();
            if (child.ExitCode != 0) throw new Exception("Declared native build failed: " + child.ExitCode);
            File.Copy(Path.Combine(source, "result.tar"), Output, true);
            foreach (var destination in Products)
                File.Copy(Path.Combine(source, "result", Path.GetFileName(destination)), destination, true);
            if (SdkArchive != null)
            {
                var archive = Path.GetFullPath(Path.Combine(source, SdkArchive));
                if (!archive.StartsWith(source + Path.DirectorySeparatorChar, StringComparison.Ordinal))
                    throw new ArgumentException("SDK archive escapes the source directory");
                Directory.CreateDirectory(SdkLayout!);
                using var compressed = File.OpenRead(archive);
                using var gzip = new GZipStream(compressed, CompressionMode.Decompress);
                TarFile.ExtractToDirectory(gzip, SdkLayout!, false);
                foreach (var name in new[] { "host", "shared", "sdk", "packs", "sdk-manifests", "templates", "library-packs", "metadata" })
                    Directory.CreateDirectory(Path.Combine(SdkLayout!, name));
            }
            return true;
        }
        finally
        {
            if (Directory.Exists(scratch)) Directory.Delete(scratch, true);
        }
    }
}
