Dictionary<string, string> Observe()
{
    var files = new ContractFiles(root, sdk);
    return Directory.EnumerateFiles(root, "*", SearchOption.AllDirectories)
        .Select(p => Path.GetRelativePath(root, p))
        .Where(p => !p.Split('/').Any(part => part is "bin" or "obj") || RestoreInput(p))
        .Order(StringComparer.Ordinal).ToDictionary(p => p, p => ContractFiles.InputDigest(files.Resolve(p)), StringComparer.Ordinal);
}
void ClearOutputs()
{
    foreach (var project in new[] { "P0", "P1", "P2" })
    {
        var bin = Path.Combine(root, project, "bin");
        if (Directory.Exists(bin)) Directory.Delete(bin, true);
        foreach (var file in Directory.EnumerateFiles(Path.Combine(root, project, "obj"), "*", SearchOption.AllDirectories).Where(p => !RestoreInput(p))) File.Delete(file);
    }
}
Dictionary<string, string> Build(bool reuse, out int evaluations, out double evaluationSeconds, out long allocated)
{
    try
    {
        ClearOutputs();
        var observed = Observe();
        var verifiedSdk = ContractFiles.TreeDigest(sdk, 4);
        if (verifiedSdk != sdkDigest) throw new InvalidDataException("SDK changed; restart evaluator with the newly declared SDK");
        var context = ContractFiles.Hash(new[] { verifiedSdk, ContractFiles.Digest(Assembly.GetExecutingAssembly().Location) }
            .Concat(Environment.GetEnvironmentVariables().Cast<System.Collections.DictionaryEntry>().OrderBy(p => (string)p.Key, StringComparer.Ordinal).Select(p => p.Key + "=" + p.Value)));
        var identity = ContractFiles.Hash(observed.Select(p => p.Key + ":" + (compilerOnly.Contains(p.Key) && !reads.Contains(p.Key) ? "reviewed compiler-only bytes" : p.Value))
            .Concat(compilerOnly.Order(StringComparer.Ordinal).Select(p => "compiler-only:" + p))
            .Concat(Directory.EnumerateFiles(sdk, "*", SearchOption.AllDirectories).Where(p => p.EndsWith(".props", StringComparison.Ordinal) || p.EndsWith(".targets", StringComparison.Ordinal)).Order(StringComparer.Ordinal).Select(p => "sdk-import-mtime:" + p + ":" + File.GetLastWriteTimeUtc(p).Ticks))
            .Concat(reads.Order(StringComparer.Ordinal).Select(p => "evaluation-read:" + p))
            .Concat(Directory.EnumerateDirectories(root, "*", SearchOption.AllDirectories).Select(p => Path.GetRelativePath(root, p)).Where(p => !p.Split('/').Any(part => part is "bin" or "obj")).Order(StringComparer.Ordinal).Select(p => "directory:" + p))
            .Concat(globals.OrderBy(p => p.Key, StringComparer.Ordinal).Select(p => p.Key + "=" + p.Value))
            .Concat(observed.Keys.Where(p => p.EndsWith(".props", StringComparison.Ordinal) || p.EndsWith(".targets", StringComparison.Ordinal) || p.EndsWith(".csproj", StringComparison.Ordinal)).Select(p => "import-mtime:" + p + ":" + File.GetLastWriteTimeUtc(Path.Combine(root, p)).Ticks)));
        session.Prepare(context, identity, reuse);
        var before = GC.GetTotalAllocatedBytes(true);
        var timer = Stopwatch.StartNew();
        var graph = new ProjectGraph([new ProjectGraphEntryPoint(Path.Combine(root, "P2/P2.csproj"), globals)], session.Collection, session.Factory);
        evaluationSeconds = timer.Elapsed.TotalSeconds;
        allocated = GC.GetTotalAllocatedBytes(true) - before;
        evaluations = session.Evaluations;
        var counter = new BuildEvaluationCounter();
        using var manager = new BuildManager();
        var result = manager.Build(new BuildParameters(session.Collection) { MaxNodeCount = 2, DisableInProcNode = true, EnableNodeReuse = false, Loggers = [counter, new ConsoleLogger(LoggerVerbosity.Minimal)] }, new GraphBuildRequestData(graph, ["Build"]));
        if (result.OverallResult != BuildResultCode.Success || counter.Evaluations != 0)
        {
            session.Clear();
            throw new InvalidDataException("Build failed or build nodes reevaluated");
        }
        if (!Observe().OrderBy(p => p.Key).SequenceEqual(observed.OrderBy(p => p.Key)))
        {
            session.Clear();
            throw new InvalidDataException("Build modified a declared input");
        }
        return Directory.EnumerateFiles(root, "*", SearchOption.AllDirectories)
            .Where(p => p.Contains("/bin/", StringComparison.Ordinal) || (p.Contains("/obj/", StringComparison.Ordinal) && p.Contains("/ref/", StringComparison.Ordinal)) || p.EndsWith("/proof.txt", StringComparison.Ordinal))
            .ToDictionary(p => Path.GetRelativePath(root, p), ContractFiles.Digest);
    }
    catch
    {
        session.Clear();
        throw;
    }
}
void Run(string label, int expected)
{
    var actual = Build(true, out var evaluated, out var seconds, out var bytes);
    if (evaluated != expected) throw new Exception($"{label}: expected {expected} evaluations, got {evaluated}");
    // Compare each build with independent fresh evaluation of these same bytes.
    // Save the retained epoch while the control uses a separate session below.
    session.Save();
    var fresh = Build(false, out var freshEvaluations, out var freshSeconds, out var freshBytes);
    session.Restore();
    if (!actual.OrderBy(p => p.Key).SequenceEqual(fresh.OrderBy(p => p.Key))) throw new Exception(label + ": product parity failed");
    rows.Add(new { label, evaluated, freshEvaluations, seconds, freshSeconds, allocatedBytes = bytes, freshAllocatedBytes = freshBytes, files = actual.Count });
    Console.WriteLine(JsonSerializer.Serialize(rows[^1]));
}

void Reject(Action action, string message)
{
    try { action(); } catch (InvalidDataException error) when (error.Message.Contains(message, StringComparison.Ordinal)) { return; }
    throw new Exception("Failure control did not reject: " + message);
}

sealed class EvaluationSession : IDisposable
{
    private string? context;
    private string? identity;
    private ProjectCollection? collection;
    private ConcurrentDictionary<string, Lazy<Project>> instances = new(StringComparer.Ordinal);
    private (ProjectCollection? Collection, ConcurrentDictionary<string, Lazy<Project>> Instances, string? Identity)? saved;
    private int evaluations;
    public ProjectCollection Collection => collection!;
    public int Evaluations => evaluations;
    public void Prepare(string nextContext, string nextIdentity, bool reuse)
    {
        if (context is not null && context != nextContext) throw new InvalidDataException("Context changed; restart evaluator");
        context = nextContext;
        if (!reuse || identity != nextIdentity)
        {
            Clear();
            identity = nextIdentity;
            collection = new ProjectCollection();
        }
        evaluations = 0;
    }
    public ProjectInstance Factory(string path, Dictionary<string, string> globals, ProjectCollection owner)
    {
        var key = path + "|" + ContractFiles.Hash(globals.OrderBy(p => p.Key, StringComparer.Ordinal).Select(p => p.Key + "=" + p.Value));
        var pristine = instances.GetOrAdd(key, _ => new Lazy<Project>(() => {
            Interlocked.Increment(ref evaluations);
            return Project.FromFile(path, new ProjectOptions { GlobalProperties = globals, ProjectCollection = owner });
        })).Value;
        // CreateProjectInstance retains item path context. DeepCopy loses that
        // context for uncached built-in metadata in this SDK's TaskItem copy.
        var evaluationId = pristine.LastEvaluationId;
        var clone = pristine.CreateProjectInstance();
        if (pristine.LastEvaluationId != evaluationId || clone.EvaluationId != evaluationId) throw new Exception("Snapshot unexpectedly reevaluated the project");
        clone.TranslateEntireState = true;
        if (clone.GetPropertyValue("BuildMutation").Length != 0 || clone.GetPropertyValue("CloneMutation").Length != 0) throw new Exception("Build state contaminated pristine evaluation");
        clone.SetProperty("CloneMutation", "one request only");
        return clone;
    }
    public void Save()
    {
        saved = (collection, instances, identity);
        collection = null;
        instances = new(StringComparer.Ordinal);
        identity = null;
    }
    public void Restore()
    {
        Clear();
        (collection, instances, identity) = saved!.Value;
        saved = null;
    }
    public void Clear()
    {
        collection?.Dispose();
        collection = null;
        instances.Clear();
        identity = null;
    }
    public void Dispose() => Clear();
}

public sealed class EvaluationProofTask : Microsoft.Build.Utilities.Task
{
    [Required] public string File { get; set; } = "";
    public string Value { get; set; } = "";
    public override bool Execute()
    {
        System.IO.Directory.CreateDirectory(Path.GetDirectoryName(File)!);
        System.IO.File.WriteAllText(File, Value);
        return true;
    }
}
